  const timeoutRef = useRef(null);

  const handleMouseEnter = (link) => {
    clearTimeout(timeoutRef.current);
    setActiveMenu(link);
  };

  const handleMouseLeave = () => {
    timeoutRef.current = setTimeout(() => setActiveMenu(null), 150);
  };

  useEffect(() => () => clearTimeout(timeoutRef.current), []);

  const MegaMenu = ({ data, isSale }) => (
    <div
      className="absolute top-full left-0 right-0 bg-white border-t-2 border-red-600 shadow-2xl z-50 px-12 py-8"
      onMouseEnter={() => clearTimeout(timeoutRef.current)}
      onMouseLeave={handleMouseLeave}
    >
      <div className="max-w-7xl mx-auto flex gap-12">
        {data.columns.map((col, i) => (
          <div key={i} className="min-w-[140px]">
            {col.heading && (
              <div className="mb-3">
                <p className="text-xs font-black tracking-widest text-black uppercase border-b border-gray-200 pb-2">
                  {col.heading}
                </p>
              </div>
            )}

                <li key={j}>
                  <a
                    href="#"
                    className={`text-sm hover:text-red-600 transition-colors duration-150 block ${
                      col.bold && !col.heading
                        ? "font-bold text-black text-base"
                        : isSale && i === 0
                        ? "font-bold text-red-600"
                        : "text-gray-600"
                    }`}
                  >
                    {item}
                  </a>
                </li>
              ))}
            </ul>
          </div>
        ))}
      </div>
    </div>
  );

          content: '';
          position: absolute;

              </button>

              </svg>

                    <div key={i} className="pt-3">
                      {col.heading && (
                        <p className="text-xs text-red-600 font-black tracking-widest uppercase mb-2">{col.heading}</p>
                      )}

                        <a key={j} href="#" className="block text-gray-400 text-sm py-1 hover:text-white transition-colors">
                          {item}
                        </a>
                      ))}
                    </div>
                  ))}
                </div>
              )}
            </div>
          ))}

          {/* Mobile Bottom Icons */}
          <div className="flex items-center gap-6 px-6 py-4 border-t border-gray-800">
            <button className="text-white text-sm font-bold tracking-wider flex items-center gap-2">
              <svg className="w-5 h-5" fill="none" stroke="currentColor" viewBox="0 0 24 24">
                <path strokeLinecap="round" strokeLinejoin="round" strokeWidth={2} d="M21 21l-4.35-4.35M17 11A6 6 0 1 1 5 11a6 6 0 0 1 12 0z" />
              </svg>
              SEARCH
            </button>
            <button className="text-white">
              <svg className="w-5 h-5" fill="none" stroke="currentColor" viewBox="0 0 24 24">
                <path strokeLinecap="round" strokeLinejoin="round" strokeWidth={2} d="M4.318 6.318a4.5 4.5 0 0 1 6.364 0L12 7.636l1.318-1.318a4.5 4.5 0 0 1 6.364 6.364L12 20.364l-7.682-7.682a4.5 4.5 0 0 1 0-6.364z" />
              </svg>
            </button>
            <button className="text-white">
              <svg className="w-5 h-5" fill="none" stroke="currentColor" viewBox="0 0 24 24">
                <path strokeLinecap="round" strokeLinejoin="round" strokeWidth={2} d="M16 7a4 4 0 1 1-8 0 4 4 0 0 1 8 0zM12 14a7 7 0 0 0-7 7h14a7 7 0 0 0-7-7z" />
              </svg>
            </button>
          </div>
        </div>
      )}

      {/* Demo Content */}
      <div className="bg-gray-100 min-h-screen flex items-center justify-center">
        <div className="text-center font-barlow">
          <p className="font-bebas text-6xl text-black tracking-widest">PLAYBOLD</p>
          <p className="text-gray-500 tracking-widest text-sm uppercase mt-2">Dare to be Different</p>
          <p className="text-gray-400 text-xs mt-4">Hover on nav links to see mega menus</p>
        </div>
      </div>
    </div>
  );
}
